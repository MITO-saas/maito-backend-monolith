1.1 Non-Negotiable Architectural PrinciplesModule Independence: Domain modules (catalog, order, payment, customer, cms) must communicate via public service interfaces (*.api.*) and immutable records. Direct cross-module entity relationships (@ManyToOne across different module domain entities) are strictly forbidden; references use raw UUIDs.Deterministic Response Wrapping: Every endpoint returns ApiResponse<T>.Auditing & Concurrency: All tables inherit from BaseAuditableEntity using UUID primary keys, UTC timestamps, and optimistic locking (@Version Long version).Zero Manual DDL: Every database change is executed via Liquibase changelogs located in src/main/resources/db/changelog/.Local Runnability: The system must build and pass all slice tests locally with a single command: ./mvnw clean test (or mvnw.cmd clean test on Windows).2.     id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    slug VARCHAR(128) UNIQUE NOT NULL,
    name VARCHAR(128) NOT NULL,
    parent_id UUID REFERENCES product_categories(id),
    description TEXT,
    is_active BOOLEAN NOT NULL DEFAULT TRUE,
    created_at TIMESTAMP WITH TIME ZONE NOT NULL DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP WITH TIME ZONE NOT NULL DEFAULT CURRENT_TIMESTAMP,

CREATE TABLE products (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    category_id UUID NOT NULL REFERENCES product_categories(id),
    sku_code VARCHAR(64) UNIQUE NOT NULL,
    name VARCHAR(255) NOT NULL,
    slug VARCHAR(255) UNIQUE NOT NULL,
    makhana_grade VARCHAR(64),
    processing_type VARCHAR(64) NOT NULL,
    hsn_code VARCHAR(16) NOT NULL DEFAULT '19041090',
    gst_rate_percent NUMERIC(4,2) NOT NULL DEFAULT 5.00,
    is_published BOOLEAN NOT NULL DEFAULT FALSE,
    created_at TIMESTAMP WITH TIME ZONE NOT NULL DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP WITH TIME ZONE NOT NULL DEFAULT CURRENT_TIMESTAMP,
    version BIGINT NOT NULL DEFAULT 0
);

CREATE TABLE product_variants (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    product_id UUID NOT NULL REFERENCES products(id) ON DELETE CASCADE,
    variant_sku VARCHAR(64) UNIQUE NOT NULL,
    flavor VARCHAR(64) DEFAULT 'CLASSIC_SALTED',
    package_weight_grams INT NOT NULL,
    uom VARCHAR(16) NOT NULL DEFAULT 'GRAMS',
    mrp_paisa BIGINT NOT NULL,
    selling_price_paisa BIGINT NOT NULL,
    wholesale_price_paisa BIGINT,
    is_active BOOLEAN NOT NULL DEFAULT TRUE,
    created_at TIMESTAMP WITH TIME ZONE NOT NULL DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP WITH TIME ZONE NOT NULL DEFAULT CURRENT_TIMESTAMP,
    version BIGINT NOT NULL DEFAULT 0
);

CREATE TABLE inventory_batches (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    variant_id UUID NOT NULL REFERENCES product_variants(id) ON DELETE RESTRICT,
    batch_number VARCHAR(64) NOT NULL,
    warehouse_code VARCHAR(32) NOT NULL DEFAULT 'WH-BIHAR-01',
    moisture_percentage NUMERIC(4,2),
    manufacturing_date DATE NOT NULL,
    expiry_date DATE NOT NULL,
    available_quantity INT NOT NULL DEFAULT 0,
    allocated_quantity INT NOT NULL DEFAULT 0,
    created_at TIMESTAMP WITH TIME ZONE NOT NULL DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP WITH TIME ZONE NOT NULL DEFAULT CURRENT_TIMESTAMP,
    version BIGINT NOT NULL DEFAULT 0,
    CONSTRAINT uk_variant_batch_wh UNIQUE (variant_id, batch_number, warehouse_code)
);

CREATE INDEX idx_products_category ON products(category_id);
CREATE INDEX idx_variants_product ON product_variants(product_id);
CREATE INDEX idx_inventory_lookup ON inventory_batches(variant_id, warehouse_code);
